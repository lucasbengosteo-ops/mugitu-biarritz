```sql
-- Un jeton secret par personne pour l'abonnement calendrier (spec admin UI § 2).
create table public.agenda_calendrier_jetons (
  user_id uuid primary key references auth.users (id) on delete cascade,
  jeton text not null unique check (char_length(jeton) >= 40),
  created_at timestamptz not null default now()
);
alter table public.agenda_calendrier_jetons enable row level security;
create policy agenda_calendrier_jetons_lecture on public.agenda_calendrier_jetons
  for select to authenticated using (user_id = auth.uid());
revoke all on public.agenda_calendrier_jetons from anon;
revoke insert, update, delete on public.agenda_calendrier_jetons from authenticated;

-- Le jeton de l'appelant, créé s'il n'existe pas ; régénéré si p_nouveau.
create or replace function public.agenda_mon_jeton_calendrier(p_nouveau boolean default false)
returns text
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_jeton text;
begin
  if auth.uid() is null or not coalesce(public.site_est_equipe(), false) then raise exception 'AGENDA_DROITS'; end if;
  if not p_nouveau then
    select jeton into v_jeton from public.agenda_calendrier_jetons where user_id = auth.uid();
    if v_jeton is not null then return v_jeton; end if;
  end if;
  v_jeton := encode(extensions.gen_random_bytes(32), 'hex');
  insert into public.agenda_calendrier_jetons (user_id, jeton) values (auth.uid(), v_jeton)
  on conflict (user_id) do update set jeton = excluded.jeton, created_at = now();
  return v_jeton;
end;
$$;
revoke all on function public.agenda_mon_jeton_calendrier(boolean) from public, anon;
grant execute on function public.agenda_mon_jeton_calendrier(boolean) to authenticated;
```
